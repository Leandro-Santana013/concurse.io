(function() {
    var processando = false;
    var liberado = false;
    function resetCaptcha() {
        try { if (typeof turnstile !== 'undefined') turnstile.reset(); } catch (e) {}
    }
    function mostraErro(msg) {
        var el = document.getElementById('turnstile-error');
        if (el) { el.style.display = 'block'; el.innerHTML = msg; }
    }
    /* escreve as URLs com hash nos links da página; o captcha continua visível */
    function aplicaLinks(arquivos) {
        var mapa = {};
        arquivos.forEach(function(a) { mapa[a.arquivo] = a; });
        document.querySelectorAll('.prova-pdf-link').forEach(function(el) {
            var item = mapa[el.dataset.arquivo];
            if (!item) return;
            if (el.dataset.acao === 'ver') {
                el.href = item.ver;
                el.target = '_blank';
            } else if (el.dataset.acao === 'baixar') {
                /* mesma página: o servidor de provas entrega como anexo */
                el.href = item.baixar;
            } else if (el.dataset.acao === 'compartilhar') {
                el.dataset.url = item.ver;
            }
        });
        liberado = true;
    }
    function liberaArquivos(token) {
        if (liberado || processando) return;
        try { if (!token && typeof turnstile !== 'undefined') token = turnstile.getResponse() || ''; } catch (e) {}
        if (!token) return;
        processando = true;
        var primeiro = document.querySelector('.prova-pdf-link');
        var dados = new FormData();
        dados.append('prova_code', primeiro ? primeiro.dataset.code : '');
        dados.append('cf-turnstile-response', token);
        fetch('https://www.pciconcursos.com.br/provas/link', { method: 'POST', body: dados })
            .then(function(r) { return r.json(); })
            .then(function(resp) {
                if (!resp.ok || !resp.arquivos) {
                    resetCaptcha();
                    mostraErro(resp.erro || 'Falha na verificação de segurança. Tente novamente.');
                    return;
                }
                /* uma verificação libera todos os arquivos da prova */
                aplicaLinks(resp.arquivos);
            })
            .catch(function() {
                resetCaptcha();
                mostraErro('Não foi possível concluir a verificação. Tente novamente.');
            })
            .finally(function() { processando = false; });
    }
    /* chamado pelo Turnstile (data-callback) ao concluir a verificação:
       libera os links de imediato, antes de qualquer clique */
    window.provasCaptchaOk = function(token) {
        if (typeof turnstileCleanError === 'function') turnstileCleanError();
        liberaArquivos(token);
    };
    document.addEventListener('click', function(e) {
        var link = e.target.closest('.prova-pdf-link');
        if (!link) return;
        if (liberado) {
            /* links liberados: navegação nativa; só o compartilhar usa JS,
               de forma síncrona dentro do gesto do usuário */
            if (link.dataset.acao === 'compartilhar') {
                e.preventDefault();
                shareLink(link.dataset.nome, 'PCI Concursos :: Provas para Download', link.dataset.url);
            }
            return;
        }
        e.preventDefault();
        if (processando) {
            mostraErro('Aguarde, liberando os arquivos...');
            return;

            liberaArquivos('');
            return;

        /* captcha ainda não resolvido: a mensagem já foi mostrada; leva até o card */
        var box = document.getElementById('captcha-provas');
        if (box) box.scrollIntoView({ behavior: 'smooth', block: 'center' });
    });
